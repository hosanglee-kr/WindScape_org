name: Deploy Application on: push: branches: - main jobs: build_and_deploy: runs-on: ubuntu-latest steps: - name: Checkout Code uses: actions/checkout@v4 - name: Create Configuration File (config.json) run: | # 1. JSON 내용을 구성합니다. # - ${{ secrets.SECRET_NAME }} 문법을 사용하여 Secret 값을 불러옵니다. # - JQ는 다중 라인 문자열을 깔끔하게 처리하기 위해 사용하거나, # 여기서는 간단한 문자열을 echo로 출력합니다. # - 환경 변수를 사용하는 것이 더 깔끔할 수 있습니다. (아래 참고) echo "{" > config.json echo " \"api_key\": \"${{ secrets.MY_API_KEY }}\"," >> config.json echo " \"db_user\": \"${{ secrets.DB_USERNAME }}\"," >> config.json echo " \"db_host\": \"${{ vars.DB_HOST }}\"" >> config.json # Variable 사용 예시 echo "}" >> config.json # 생성된 파일 내용 확인 (디버깅용, 실제 키는 마스킹 처리됨) cat config.json # 3. 이후 빌드/배포 스텝에서 생성된 config.json 파일을 사용합니다. - name: Deploy with new configuration run: | # npm install # node deploy-script.js --config config.json echo "Configuration file ready for deployment."
